import heapq
# CS 124 Dijkstra Min-Heap Implementation with State Invalidation
def dijkstra(graph, start_vertex):
    distances = {v: float('inf') for v in graph}
    distances[start_vertex] = 0
    # Tuple: (current_cost, vertex_id)
    min_pq = [(0, start_vertex)]

    while min_pq:
        curr_dist, u = heapq.heappop(min_pq)
        if curr_dist > distances[u]:
            continue  # Lazy deletion of stale heap states

        for neighbor, weight in graph[u].items():
            new_dist = curr_dist + weight
            if new_dist < distances[neighbor]:
                distances[neighbor] = new_dist
                heapq.heappush(min_pq, (new_dist, neighbor))
    return distances
